```python
# ----------------------------------------------------------------------------------- CLAUDE.md (0.33.0)
# Claude Code (перевірено 20.11.2026 на 2.1.322) не читає AGENTS.md, коли вище по теках є будь-який CLAUDE.md; CLAUDE.md у теці
# простору з рядком `@AGENTS.md` це лагодить. Наявний CLAUDE.md змінює ЛИШЕ `claude_connect` (кнопка «Підключити» = згода людини).

CLAUDE_NAME = 'CLAUDE.md'
_IMPORT_RE = re.compile(r'^[ \t]*@(?:\./)?AGENTS\.md[ \t]*\r?$', re.M)
_IMPORT_LINE = '@AGENTS.md\n'


def _claude_path(space_dir):
    return os.path.join(space_dir, CLAUDE_NAME)


def _claude_read(space_dir):
    """Текст CLAUDE.md чи `None` (файла нема). Симлінк, не-звичайний файл, відмова читання — `OSError` (як `_read`)."""
    p = _claude_path(space_dir)
    if os.path.islink(p) or (os.path.lexists(p) and not os.path.isfile(p)):
        raise OSError(f'{CLAUDE_NAME}: not a regular file')
    try:
        with open(p, **_ENC) as f:
            return f.read()
    except FileNotFoundError:
        return None


def claude_status(space_dir):
    """`{state, path}`: `ok` (є рядок `@AGENTS.md` чи `@./AGENTS.md`), `missing`, `no-import`, `unreadable`. Не кидає."""
    p = _claude_path(space_dir)
    try:
        text = _claude_read(space_dir)
    except (OSError, ValueError):
        return {'state': 'unreadable', 'path': p}
    if text is None:
        return {'state': 'missing', 'path': p}
    return {'state': 'ok' if _IMPORT_RE.search(text) else 'no-import', 'path': p}


def claude_ensure(space_dir):
    """Новий простір і «Створити» AGENTS.md: CLAUDE.md з `@AGENTS.md` ЛИШЕ коли його нема (`O_EXCL`); наявний не чіпаємо.
    `True` — створено."""
    try:
        with open(_claude_path(space_dir), 'x', **_ENC) as f:
            f.write(_IMPORT_LINE)
        return True
    except FileExistsError:
        return False


def claude_connect(space_dir):
    """«Підключити»: нема файла — створити з `@AGENTS.md`; є без імпорту — дописати `@AGENTS.md` і порожній рядок НА ПОЧАТОК
    (решта байт у байт, права ті самі, запис атомарний); уже підключено — нічого. Нечитний — `OSError`; файл змінили між читанням
    і заміною — `ValueError('changed')`. Повертає `claude_status` після запису."""
    text = _claude_read(space_dir)
    if text is None:
        if not claude_ensure(space_dir):
            raise ValueError('changed')          # файл зʼявився між читанням і створенням
        return claude_status(space_dir)
    if _IMPORT_RE.search(text):
        return claude_status(space_dir)
    path = _claude_path(space_dir)
    mode = os.stat(path).st_mode & 0o777
    tmp = tempfile.NamedTemporaryFile('w', dir=space_dir, prefix='.CLAUDE.', suffix='.tmp', delete=False, **_ENC)
    try:
        with tmp:
            tmp.write(_IMPORT_LINE + '\n' + text)
            tmp.flush()
            os.fsync(tmp.fileno())
        os.chmod(tmp.name, mode)
        if _claude_read(space_dir) != text:      # чужу правку не губимо
            raise ValueError('changed')
        os.replace(tmp.name, path)
    except BaseException:
        try:
            os.unlink(tmp.name)
        except OSError:
            pass
        raise
    return claude_status(space_dir)
```

```js
/**
 * Rebuilds the demo task template (0.33.0): `app/templates/demo/` (pages, `pages.json`, `flow.json`) and
 * `app/templates/demo-pins.json` — two pins with anchor fingerprints computed in Chromium by the same
 * `CommentLayerAnchor.fingerprint` the commenter uses (`app/vendor/anchor.min.js`). For developers, not for the runtime:
 * `POST /api/demo-task` only copies these files. Needs Playwright's Chromium (`npx playwright install chromium`).

 *   node ops/demo-task.mjs
 *
 * Files are overwritten, never deleted: a page that left `demo-pages.mjs` stays on disk until removed by hand, and
 * `tests/demo-task.test.mjs` (directory listing == `demoFiles()`) says so.
 */
import { mkdirSync, readFileSync, writeFileSync } from "node:fs";
import path from "node:path";
import { fileURLToPath } from "node:url";
import { chromium } from "playwright";
import { DEMO_PINS } from "./readme-shots/demo-pages.mjs";
import { DEMO_VERSION, demoFiles } from "./demo-task/build.mjs";

const ROOT = path.resolve(path.dirname(fileURLToPath(import.meta.url)), "..");
const OUT = path.join(ROOT, "app", "templates", "demo");
const PINS_OUT = path.join(ROOT, "app", "templates", "demo-pins.json");
const ANCHOR = readFileSync(path.join(ROOT, "app", "vendor", "anchor.min.js"), "utf8");

const files = demoFiles();
for (const [rel, text] of Object.entries(files)) {
  mkdirSync(path.dirname(path.join(OUT, rel)), { recursive: true });
  writeFileSync(path.join(OUT, rel), text);
}

const browser = await chromium.launch();
try {
  const page = await browser.newPage({ viewport: { width: 1440, height: 900 } });
  const pins = [];
  for (const p of DEMO_PINS) {
    await page.setContent(files[`${p.page}/index.html`]);
    await page.addScriptTag({ content: ANCHOR });
    const fp = await page.evaluate((s) => {
      const el = document.querySelector(s);
      return el ? window.CommentLayerAnchor.fingerprint(el, document.body) : null;
    }, p.sel);
    if (!fp) throw new Error(`no pin target ${p.page} ${p.sel}`);
    pins.push({ ...p, fp });
  }
  writeFileSync(PINS_OUT, `${JSON.stringify({ version: DEMO_VERSION, pins }, null, 1)}\n`);
} finally {
  await browser.close();
}
```

````markdown
## Steps

### 1. Check the computer

`git --version`; `python3 --version` (3.9 or newer); Google Chrome or Microsoft Edge is installed; `node --version` (20 or
newer). Node is optional: without it the Overview screenshots and the "System Vercel CLI" login do not work. If something
required is missing, tell the person what and how to install it, and ask before doing it.

### 2. Get the code

`git clone https://github.com/neodisa/deai-canvas.git ~/deai-canvas` — or into the folder the person named. If the folder
already exists and is a Deai Canvas checkout, use it as it is. Below, `<deai>` is that folder.

### 3. Screenshots for Overview

In `<deai>`: `npm ci && npx playwright install chromium`. No Node — tell the person and go on.

### 4. App icon and autostart (macOS)

`bash <deai>/ops/canvas-app/install.sh`, then check it: `bash <deai>/ops/canvas-app/install.sh --check`.

### 5. Start

Run `<deai>/start.command`. Wait until `GET http://127.0.0.1:8765/api/setup` answers (up to 30 seconds).

### 6. Check

`GET /api/setup` → `shots.state` should be `ok`. Otherwise tell the person why: `no-node` — Node.js 20+ is needed;
`no-playwright` or `no-browser` — run `shots.command`; `unknown` — the check itself failed, go on.

### 7. Ask once

Send ONE message with four questions; each has a default the person can accept with one word:

1. Space name and folder — default: "Personal", `~/deai-work/personal`.
2. Where to publish previews — a folder on this computer (default) or their own Vercel.
3. One password for the previews of all tasks in the space — default: no password.
4. A git repository for the mockups — default: later.

### 8. Create the space

`POST /api/space-new {"name": "…", "color": "space-1", "path": "…"}` → `{ok, id}`. The canvas puts `AGENTS.md` and
`CLAUDE.md` (the line `@AGENTS.md`, so Claude Code reads `AGENTS.md`) into the folder. If the folder already is a space,
the canvas picks it up.

### 9. Modules

The fields of every module: `GET /api/modules?space=<id>`.

- Publishing to a folder: `POST /api/module-enable {"space": "<id>", "id": "static-publish"}`; the password —
  `POST /api/module-settings {"space": "<id>", "id": "static-publish", "secrets": {"password": "…"}}`.
- Publishing to Vercel: `POST /api/module-enable {"space": "<id>", "id": "vercel-publish"}`; settings and the password —
  `POST /api/module-settings`. The person can type the Vercel token in themselves: Settings → Modules → Vercel.
- Git: `git-sync` with the repository the person gave — `module-enable`, then `module-settings`.

After each module: `POST /api/module-check {"space": "<id>", "id": "…"}`, and tell the person the result.

### 10. Demo task

`POST /api/demo-task {"space": "<id>"}` → `{ok, id}`: a made-up budgeting app "Tally" with five pages, two states, a flow
and two pins — to try Overview, comments and the inspector without an agent.

### 11. Open the canvas

If the icon is installed: `open -a "Design Canvas"`. Otherwise: `open http://localhost:8765/app/`.

### 12. The last message

Tell the person:

- what is done and what is not (with the reason);
- what is where, in 5–6 lines: task tabs at the top; the bar with Layout, Overview, Flow and Publish; the page tree on the
  left; the island at the bottom with comments (C), the inspector (I) and bug or idea; ⌘K for every action;
- how to work next: `cd <space folder> && claude`, then phrases like "make a payment screen" or "work through the comments";
- ⌘K → "Canvas tour" shows the map again.

## Agent status

The island shows what the agent is doing only from a heartbeat: `POST /api/claude-status {"task", "phase", "status",
"note", "session"}`. Without it everything else works. Add hooks that send it to the person's Claude Code only if the
person asks for it.
````

```js
/** Блок «Кадри» за `shots` з `GET /api/setup`: `{lineKey, whyKey, command}` — знімати нема чим; `null` — можна (ok,
    unknown, ще не питали). `no-playwright` і `no-browser` лікує одна й та сама команда, тож і текст один. */
export function shotsBlock(setupShots) {
  const st = setupShots?.state;
  if (st === "no-playwright" || st === "no-browser") {
    return { lineKey: "frShotsNoBrowser", whyKey: "frShotsNoBrowserWhy", command: setupShots.command || "" };
  }
  if (st === "no-node") return { lineKey: "frShotsNoNode", whyKey: "frShotsNoNodeWhy", command: "" };
  return null;
}

/** Рядок «Claude Code» розділу AGENTS.md за `claude` з `GET /api/agents-md`; `null` — стану нема (старий сервер). */
export function claudeRow(claude) {
  switch (claude?.state) {
    case "ok": return { icon: "check", tone: "ok", textKey: "frClaudeOk", connect: false };
    case "missing": return { icon: "warn", tone: "warn", textKey: "frClaudeMissing", connect: true };
    case "no-import": return { icon: "warn", tone: "warn", textKey: "frClaudeNoImport", connect: true };
    case "unreadable": return { icon: "warn", tone: "warn", textKey: "frClaudeUnreadable", connect: false };
    default: return null;
  }
}

/** Шлях для shell: прості символи — як є, інакше в одинарних лапках (`'` → `'\''`). */
export const shq = (s) => (/^[A-Za-z0-9_@%+=:,./~-]+$/.test(s) ? s : `'${String(s).replace(/'/g, "'\\''")}'`);

/** «Запуск агента»: `cd <тека простору> && claude` із шляху AGENTS.md; `""` — шляху нема. */
export function runCommand(agentsPath) {
  if (typeof agentsPath !== "string" || !agentsPath) return "";
  const i = agentsPath.lastIndexOf("/");
  const dir = i > 0 ? agentsPath.slice(0, i) : "/";
  return `cd ${shq(dir)} && claude`;
}

/** Рядок пароля на картці модуля: увімкнений, наявний постачальник ролі `publish` з налаштуванням `password` типу `secret`.
    `{state: "set"|"unset"|"unknown"}` або `null`. */
export function pwLine(e) {
  if (!e || e.enabled !== true || e.status === "missing" || !Array.isArray(e.roles) || !e.roles.includes("publish")) return null;
  const f = (Array.isArray(e.settings) ? e.settings : []).find((s) => s?.key === "password" && s.type === "secret");
  if (!f) return null;
  return { state: f.set === true ? "set" : f.set === false ? "unset" : "unknown" };
}

/** Пункт «Пароль превʼю» меню публікації: поле простору `publish_password` (`true|false|null`); поля нема — пункту нема. */
export function pwItem(space) {
  if (!space || !Object.prototype.hasOwnProperty.call(space, "publish_password")) return null;
  const v = space.publish_password;
  return { state: v === true ? "set" : v === false ? "unset" : "unknown" };
}
```
